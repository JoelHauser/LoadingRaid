# Deploy Screen -- which backdrop each map gets.
#
# Only read when "Match the map" is on in the F12 settings. Anything not named here
# keeps the built-in choice; delete a line to go back to it.
#
#   <location id> = <backdrop>
#
# Backdrops:   Factory  Wood  Laboratory  TheUnheardEdition  Cyber  Random
#
# Random means "leave the menu alone for this map".
#
# TheUnheardEdition and Cyber are edition-themed and are not in every install. If one
# is missing the mod logs it and leaves the backdrop alone rather than loading nothing.
#
# Location ids are matched without regard to case. They are the "Id" field inside each
# map's base.json -- not the folder name, which is lowercase for every map:
#
#   bigmap (Customs)      factory4_day        factory4_night      laboratory
#   Interchange           Labyrinth           Lighthouse          RezervBase (Reserve)
#   Sandbox (Ground Zero) Sandbox_high        Shoreline           Suburbs
#   TarkovStreets         Terminal            Town                Woods
#
# The defaults, for reference -- uncomment and edit any of them:
#
# factory4_day   = Factory
# factory4_night = Factory
# bigmap         = Factory
# Interchange    = Factory
# RezervBase     = Factory
# TarkovStreets  = Factory
# laboratory     = Laboratory
# Labyrinth      = Laboratory
# Woods          = Wood
# Shoreline      = Wood
# Lighthouse     = Wood
# Sandbox        = Wood
# Sandbox_high   = Wood
# Suburbs        = Wood
# Town           = Wood
# Terminal       = Wood
